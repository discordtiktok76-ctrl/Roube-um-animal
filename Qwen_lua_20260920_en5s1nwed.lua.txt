--// Helpers -----------------------------------------------------------------
local function safeGet(obj, prop)
	local ok, value = pcall(function() return obj[prop] end)
	return ok and value or nil
end

local function safeSet(obj, prop, value)
	return pcall(function() obj[prop] = value end)
end

local function clamp(n, a, b)
	return math.max(a, math.min(b, n))
end

local function round(n)
	return math.floor(n + 0.5)
end

local function clearTable(t)
	for k in pairs(t) do t[k] = nil end
end

local function splitEnumString(str)
	local enumType, enumName = string.match(tostring(str), "^Enum%.([%w_]+)%.([%w_]+)$")
	return enumType, enumName
end

local function getExecutorEnv()
	if type(getgenv) == "function" then
		local ok, env = pcall(getgenv)
		if ok and type(env) == "table" then return env end
	end
	return _G
end

local function safeClipboard(text)
	local env = getExecutorEnv()
	for _, name in ipairs({"setclipboard", "toclipboard", "set_clipboard"}) do
		local fn = rawget(env, name)
		if type(fn) == "function" then
			local ok = pcall(fn, text)
			if ok then return true end
		end
	end
	return false
end

local function getRequestFunction()
	local env = getExecutorEnv()
	for _, name in ipairs({"request", "http_request", "syn_request"}) do
		local fn = rawget(env, name)
		if type(fn) == "function" then return fn end
	end
	return nil
end

local function urlEncode(s)
	s = tostring(s or "")
	return s:gsub("([^%w%-_%.~])", function(c)
		return string.format("%%%02X", string.byte(c))
	end)
end

local function make(className, props, parent)
	local ok, obj = pcall(Instance.new, className)
	if not ok or not obj then return nil end
	for k, v in pairs(props or {}) do
		pcall(function() obj[k] = v end)
	end
	if parent then obj.Parent = parent end
	return obj
end

local function tween(obj, props, duration)
	if not obj then return end
	local ok, info = pcall(TweenInfo.new, duration or 0.13, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
	if not ok then return end
	local ok2, tw = pcall(function() return TweenService:Create(obj, info, props) end)
	if ok2 and tw then tw:Play() end
end

local function corner(obj, radius)
	if not obj then return end
	local c = obj:FindFirstChild("NORLI_Corner") or obj:FindFirstChildOfClass("UICorner")
	if not c then
		c = Instance.new("UICorner")
		c.Name = "NORLI_Corner"
		c.Parent = obj
	end
	c.CornerRadius = UDim.new(0, radius or 10)
	return c
end

local function getStroke(obj)
	if not obj then return nil end
	return obj:FindFirstChild("NORLI_Stroke") or obj:FindFirstChildOfClass("UIStroke")
end

local function ensureStroke(obj)
	if not obj then return nil end
	local s = getStroke(obj)
	if not s then
		s = Instance.new("UIStroke")
		s.Name = "NORLI_Stroke"
		s.Parent = obj
	end
	return s
end

local function removeUiStroke(obj)
	local s = getStroke(obj)
	if s and s.Name ~= "NORLI_Selection" then
		pcall(function() s:Destroy() end)
	end
end

local function newId()
	local ok, guid = pcall(function() return HttpService:GenerateGUID(false) end)
	if ok and guid then
		return tostring(guid):gsub("-", "")
	end
	return "ID" .. tostring(math.random(100000, 999999)) .. tostring(os.clock()):gsub("%.", "")
end

local function toast(message, color)
	toastToken += 1
	local token = toastToken

	Toast.Text = tostring(message or "")
	Toast.BackgroundColor3 = color or C.card
	Toast.Visible = true
	Toast.Position = UDim2.new(0.5, 0, 1, -106)
	tween(Toast, {Position = UDim2.new(0.5, 0, 1, -120)}, 0.12)

		if token ~= toastToken then return end
		tween(Toast, {Position = UDim2.new(0.5, 0, 1, -92)}, 0.12)
		task.delay(0.14, function()
			if token == toastToken then Toast.Visible = false end
		end)
	end)

--// Utility object checks ----------------------------------------------------
local function isUserElement(obj)
	return obj ~= nil
		and obj:IsA("GuiObject")
		and obj ~= EmptyHint
		and obj.Name ~= "NORLI_Selection"
end

local function isSelectable(obj)
	return obj ~= nil
		and obj:IsA("GuiObject")
		and obj:IsDescendantOf(DesignRoot)
		and obj ~= DesignRoot
		and obj ~= EmptyHint
		and obj.Name ~= "NORLI_Selection"
end

local function isLocked(obj)
	return obj ~= nil and obj:GetAttribute("NORLI_Locked") == true
end

local function updateEmpty()
	local count = 0
	for _, child in ipairs(DesignRoot:GetChildren()) do
		if isUserElement(child) then count += 1 end
	end
	EmptyHint.Visible = count == 0
end

local function findObjectById(id)
	if not id then return nil end
	for _, descendant in ipairs(DesignRoot:GetDescendants()) do
		if descendant:IsA("GuiObject") and descendant:GetAttribute("NORLI_ID") == id then
			return descendant
		end
	end
	return nil
end

local function removeEditorHelpersRecursive(obj)
	if not obj then return end
	local destroyList = {}
	for _, descendant in ipairs(obj:GetDescendants()) do
		if descendant.Name == "NORLI_Selection" then
			table.insert(destroyList, descendant)
		end
	end
	for _, item in ipairs(destroyList) do
		pcall(function() item:Destroy() end)
	end
end

local function ensureId(obj)
	if obj and obj:IsA("GuiObject") and not obj:GetAttribute("NORLI_ID") then
		obj:SetAttribute("NORLI_ID", newId())
	end
end

local function ensureTreeIds(root, forceNew)
	local function assign(obj)
		if obj:IsA("GuiObject") then
			if forceNew or not obj:GetAttribute("NORLI_ID") then
				obj:SetAttribute("NORLI_ID", newId())
			end
		end
	end

	if root then assign(root) end
	for _, descendant in ipairs(root:GetDescendants()) do
		assign(descendant)
	end
end

local function setSelected(obj)
	if obj == EmptyHint then obj = nil end
	if obj and not isSelectable(obj) then obj = nil end

	selected = obj
	removeSelectionVisual()

	if selected then
		showSelectionVisual()
	end

	updateEmpty()
	refreshProperties()

	if currentTab == "LAYERS" then
		refreshLayers()
	end
end

--// Connect selectable ------------------------------------------------------
connectSelectable = function(obj)
	if not obj or not obj:IsA("GuiObject") then return end
	if obj == EmptyHint or obj.Name == "NORLI_Selection" then return end
	if obj:GetAttribute("NORLI_Connected") == EDITOR_TOKEN then return end

	obj:SetAttribute("NORLI_Connected", EDITOR_TOKEN)
	obj.Active = true

	obj.InputBegan:Connect(function(input)
		if panel then return end
		if isLocked(obj) then return end
		if input.Target ~= obj then return end

		if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
			if obj:IsA("TextBox") then
				local ok, focused = pcall(function() return obj:IsFocused() end)
				if ok and focused then return end
			end

			setSelected(obj)
			beginDrag(obj, input)
		end
	end)
end

prepareTree = function(root, forceNewIds)
	if not root then return end

	removeEditorHelpersRecursive(root)

	local function clearToken(obj)
		if obj:IsA("GuiObject") then
			obj:SetAttribute("NORLI_Connected", nil)
		end
	end

	clearToken(root)
	for _, descendant in ipairs(root:GetDescendants()) do
		clearToken(descendant)
	end

	ensureTreeIds(root, forceNewIds == true)

	if root:IsA("GuiObject") then
		connectSelectable(root)
	end

	for _, descendant in ipairs(root:GetDescendants()) do
		if descendant:IsA("GuiObject") then
			connectSelectable(descendant)
		end
	end
end

--// History -----------------------------------------------------------------
local MAX_HISTORY = 40
local undoStack = {}
local redoStack = {}

local function captureState()
	local visualExisted = selectionVisual ~= nil
	removeSelectionVisual()

	local items = {}
	for _, child in ipairs(DesignRoot:GetChildren()) do
		if isUserElement(child) then
			table.insert(items, child:Clone())
		end
	end

	local selectedId = selected and selected:GetAttribute("NORLI_ID") or nil

	if selected and visualExisted then
		showSelectionVisual()
	end

	return {
		items = items,
		selectedId = selectedId,
	}
end

local function pushPreState(state)
	if suppressHistory or restoring then return end
	table.insert(undoStack, state)
	if #undoStack > MAX_HISTORY then
		table.remove(undoStack, 1)
	end
	clearTable(redoStack)
end

local function saveUndoPoint()
	if restoring or suppressHistory then return end
	pushPreState(captureState())
end

local function clearUserElements()
	for _, child in ipairs(DesignRoot:GetChildren()) do
		if isUserElement(child) then
			pcall(function() child:Destroy() end)
		end
	end
end

local function restoreState(state)
	restoring = true

	removeSelectionVisual()
	clearUserElements()

	for _, instance in ipairs(state.items or {}) do
		local clone = instance:Clone()
		clone.Parent = DesignRoot
		prepareTree(clone, false)
	end

	updateEmpty()

	local target = state.selectedId and findObjectById(state.selectedId) or nil
	if target then
		setSelected(target)
	else
		setSelected(nil)
	end

	restoring = false

	if currentTab == "LAYERS" then refreshLayers() end
	if currentTab == "INSPECT" then refreshProperties() end
end

doUndo = function()
	if #undoStack == 0 then
		toast("NADA PARA DESFAZER", C.muted)
		return
	end

	local current = captureState()
	local state = table.remove(undoStack)
	table.insert(redoStack, current)
	restoreState(state)
	toast("DESFEITO", C.accent)
end

doRedo = function()
	if #redoStack == 0 then
		toast("NADA PARA REFAZER", C.muted)
		return
	end

	local current = captureState()
	local state = table.remove(redoStack)
	table.insert(undoStack, current)
	restoreState(state)
	toast("REFEITO", C.accent)
end

--// Input helpers -----------------------------------------------------------
local function getStageScale()
	return ZoomScale.Scale
end

local function sameInput(inp, origin)
	if inp == origin then return true end
	if origin.UserInputType == Enum.UserInputType.MouseButton1 and inp.UserInputType == Enum.UserInputType.MouseMovement then
		return true
	end
	if origin.UserInputType == Enum.UserInputType.Touch and inp.UserInputType == Enum.UserInputType.Touch then
		return true
	end
	return false
end

beginDrag = function(obj, input)
	if not obj or isLocked(obj) then return end

	local preState = captureState()
	local startInputPos = input.Position
	local startObjPos = obj.Position
	local moved = false

	local conn1, conn2

	conn1 = UserInputService.InputChanged:Connect(function(inp)
		if not sameInput(inp, input) then return end

		local delta = (inp.Position - startInputPos) / getStageScale()
		if delta.Magnitude > 2 then moved = true end

		local newX = startObjPos.X.Offset + delta.X
		local newY = startObjPos.Y.Offset + delta.Y

		if snapEnabled then
			newX = round(newX / gridSize) * gridSize
			newY = round(newY / gridSize) * gridSize
		end

		obj.Position = UDim2.new(startObjPos.X.Scale, newX, startObjPos.Y.Scale, newY)

		if obj == selected then
			showSelectionVisual()
		end
	end)

	conn2 = UserInputService.InputEnded:Connect(function(inp)
		if not sameInput(inp, input) then return end
		pcall(function() conn1:Disconnect() end)
		pcall(function() conn2:Disconnect() end)

		if moved then
			pushPreState(preState)
		end
	end)
end

beginResize = function(obj, input, hx, hy)
	if not obj or isLocked(obj) then return end

	local preState = captureState()
	local startSize = obj.Size
	local startPos = obj.Position
	local startInputPos = input.Position

	local rad = math.rad(obj.Rotation)
	local cosA = math.cos(rad)
	local sinA = math.sin(rad)

	local conn1, conn2

	conn1 = UserInputService.InputChanged:Connect(function(inp)
		if not sameInput(inp, input) then return end

		local raw = (inp.Position - startInputPos) / getStageScale()
		local localDX = raw.X * cosA + raw.Y * sinA
		local localDY = -raw.X * sinA + raw.Y * cosA

		local newW = startSize.X.Offset + localDX * hx
		local newH = startSize.Y.Offset + localDY * hy

		newW = clamp(newW, 16, 3000)
		newH = clamp(newH, 16, 3000)

		if snapEnabled then
			newW = round(newW / gridSize) * gridSize
			newH = round(newH / gridSize) * gridSize
		end

		local adjX = hx == -1 and (startSize.X.Offset - newW) or 0
		local adjY = hy == -1 and (startSize.Y.Offset - newH) or 0

		local worldAdjX = adjX * cosA - adjY * sinA
		local worldAdjY = adjX * sinA + adjY * cosA

		obj.Size = UDim2.new(startSize.X.Scale, newW, startSize.Y.Scale, newH)
		obj.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + worldAdjX,
			startPos.Y.Scale,
			startPos.Y.Offset + worldAdjY
		)

		if obj == selected then
			showSelectionVisual()
		end
	end)

	conn2 = UserInputService.InputEnded:Connect(function(inp)
		if not sameInput(inp, input) then return end
		pcall(function() conn1:Disconnect() end)
		pcall(function() conn2:Disconnect() end)

		pushPreState(preState)
	end)
end

beginRotate = function(obj, input)
	if not obj or isLocked(obj) then return end

	local preState = captureState()
	local startRotation = obj.Rotation
	local center = obj.AbsolutePosition + obj.AbsoluteSize / 2
	local startAngle = math.deg(math.atan2(input.Position.Y - center.Y, input.Position.X - center.X))

	local conn1, conn2

	conn1 = UserInputService.InputChanged:Connect(function(inp)
		if not sameInput(inp, input) then return end

		local angle = math.deg(math.atan2(inp.Position.Y - center.Y, inp.Position.X - center.X))
		local newRotation = startRotation + (angle - startAngle)

		if snapEnabled then
			newRotation = round(newRotation / 15) * 15
		end

		obj.Rotation = newRotation

		if obj == selected then
			showSelectionVisual()
		end
	end)

	conn2 = UserInputService.InputEnded:Connect(function(inp)
		if not sameInput(inp, input) then return end
		pcall(function() conn1:Disconnect() end)
		pcall(function() conn2:Disconnect() end)

		pushPreState(preState)
	end)
end

local function uniqueName(base, parent)
	local n = 1
	local name = base
	while parent and parent:FindFirstChild(name) do
		n += 1
		name = base .. n
	end
	return name
end

local function rememberRecent(className)
	for i = #recentClasses, 1, -1 do
		if recentClasses[i] == className then
			table.remove(recentClasses, i)
		end
	end

	table.insert(recentClasses, 1, className)

	while #recentClasses > 5 do
		table.remove(recentClasses)
	end
end

local function createElement(className)
	saveUndoPoint()

	local parentObj = DesignRoot
	if selected and selected:IsA("GuiObject") and selected ~= EmptyHint and selected ~= DesignRoot and not isLocked(selected) then
		parentObj = selected
	end

	local ok, obj = pcall(Instance.new, className)
	if not ok or not obj then
		toast("ERRO AO CRIAR ELEMENTO", C.red)
		return nil
	end

	obj.Name = uniqueName(className, parentObj)

	local props = defaultProps[className] and defaultProps[className]() or {}
	for k, v in pairs(props) do
		safeSet(obj, k, v)
	end

	obj.Position = UDim2.fromOffset(20, 20)
	obj.ZIndex = clamp((parentObj ~= DesignRoot and parentObj.ZIndex or 5) + 1, 1, 999)

	if className ~= "TextLabel" then
		corner(obj, 10)
	else
		corner(obj, 6)
	end

	ensureId(obj)
	obj.Parent = parentObj
	connectSelectable(obj)

	setSelected(obj)
	rememberRecent(className)
	updateEmpty()
	toast(className:upper() .. " CRIADO", C.green)

	return obj
end

local function alignSelected(mode)
	if not selected then
		toast("SELECIONE UM ELEMENTO", C.muted)
		return
	end

	saveUndoPoint()

	local parentSize = getParentSize(selected)
	local size = selected.AbsoluteSize
	local ax = selected.AnchorPoint.X
	local ay = selected.AnchorPoint.Y
	local pos = selected.Position

	local function setX(x)
		selected.Position = UDim2.new(pos.X.Scale, x, pos.Y.Scale, selected.Position.Y.Offset)
	end

	local function setY(y)
		selected.Position = UDim2.new(selected.Position.X.Scale, selected.Position.X.Offset, pos.Y.Scale, y)
	end

	if mode == "left" then
		setX(size.X * ax)
	elseif mode == "hcenter" then
		setX((parentSize.X - size.X) / 2 + size.X * ax)
	elseif mode == "right" then
		setX(parentSize.X - size.X + size.X * ax)
	elseif mode == "top" then
		setY(size.Y * ay)
	elseif mode == "vcenter" then
		setY((parentSize.Y - size.Y) / 2 + size.Y * ay)
	elseif mode == "bottom" then
		setY(parentSize.Y - size.Y + size.Y * ay)
	elseif mode == "fill" then
		selected.Size = UDim2.new(1, 0, 1, 0)
		selected.Position = UDim2.new(selected.AnchorPoint.X, 0, selected.AnchorPoint.Y, 0)
	elseif mode == "fillw" then
		selected.Size = UDim2.new(1, 0, selected.Size.Y.Scale, selected.Size.Y.Offset)
	elseif mode == "fillh" then
		selected.Size = UDim2.new(selected.Size.X.Scale, selected.Size.X.Offset, 1, 0)
	end

	showSelectionVisual()
	toast("ALINHADO", C.accent)
end

local function luaString(v)
	v = tostring(v or "")
	v = v:gsub("\\", "\\\\")
	v = v:gsub("\n", "\\n")
	v = v:gsub("\r", "\\r")
	v = v:gsub("\"", "\\\"")
	return "\"" .. v .. "\""
end

local function nstr(n)
	n = tonumber(n) or 0
	if math.abs(n) < 0.000001 then n = 0 end
	local s = string.format("%.4f", n)
	s = s:gsub("(%..-)0+$", "%1"):gsub("%.$", "")
	return s
end

local function ser(v)
	local t = typeof(v)

	if t == "string" then return luaString(v) end
	if t == "number" then return nstr(v) end
	if t == "boolean" then return v and "true" or "false" end
	if t == "Color3" then return string.format("Color3.fromRGB(%d,%d,%d)", round(v.R * 255), round(v.G * 255), round(v.B * 255)) end
	if t == "UDim" then return string.format("UDim.new(%s,%s)", nstr(v.Scale), nstr(v.Offset)) end
	if t == "UDim2" then return string.format("UDim2.new(%s,%s,%s,%s)", nstr(v.X.Scale), nstr(v.X.Offset), nstr(v.Y.Scale), nstr(v.Y.Offset)) end
	if t == "Vector2" then return string.format("Vector2.new(%s,%s)", nstr(v.X), nstr(v.Y)) end
	if t == "Vector3" then return string.format("Vector3.new(%s,%s,%s)", nstr(v.X), nstr(v.Y), nstr(v.Z)) end
	if t == "EnumItem" then return tostring(v) end
	if t == "BrickColor" then return "BrickColor.new(" .. luaString(v.Name) .. ")" end
	if t == "Rect" then return string.format("Rect.new(%s,%s,%s,%s)", nstr(v.Min.X), nstr(v.Min.Y), nstr(v.Max.X), nstr(v.Max.Y)) end

	return nil
end

local function shouldSkipLua(obj)
	if obj.Name == "NORLI_Selection" then return true end
	if obj == EmptyHint then return true end

	if not (
		obj:IsA("GuiObject") or obj:IsA("UICorner") or obj:IsA("UIStroke") or
		obj:IsA("UIGradient") or obj:IsA("UIPadding") or obj:IsA("UIAspectRatioConstraint") or
		obj:IsA("UISizeConstraint") or obj:IsA("UIScale") or obj:IsA("UIListLayout") or
		obj:IsA("UIGridLayout") or obj:IsA("UIPageLayout")
	) then
		return true
	end

	return false
end

local function appendProps(lines, var, obj, props)
	for _, prop in ipairs(props) do
		local value = safeGet(obj, prop)
		local encoded = value ~= nil and ser(value)
		if encoded then
			table.insert(lines, var .. "." .. prop .. " = " .. encoded)
		end
	end
end

local function serializeTree(lines, obj, parentExpr, vars, indexRef)
	if shouldSkipLua(obj) then return end

	indexRef.n += 1
	local var = "obj" .. indexRef.n
	vars[obj] = var

	table.insert(lines, "local " .. var .. " = Instance.new(" .. luaString(obj.ClassName) .. ")")

	if obj:IsA("GuiObject") then
		appendProps(lines, var, obj, exportGuiProps)

		if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
			appendProps(lines, var, obj, exportTextProps)
		end

		if obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
			appendProps(lines, var, obj, exportImageProps)
		end

		if obj:IsA("ScrollingFrame") then
			appendProps(lines, var, obj, exportScrollProps)
		end

		if obj:IsA("VideoFrame") then
			appendProps(lines, var, obj, exportVideoProps)
		end

		if obj:IsA("TextButton") or obj:IsA("ImageButton") then
			local interact = safeGet(obj, "Interactable")
			if interact ~= nil then
				table.insert(lines, var .. ".Interactable = " .. ser(interact))
			end
		end
	elseif obj:IsA("UIStroke") then
		appendProps(lines, var, obj, exportStrokeProps)
	elseif obj:IsA("UICorner") then
		appendProps(lines, var, obj, exportCornerProps)
	elseif obj:IsA("UIPadding") then
		appendProps(lines, var, obj, exportPaddingProps)
	elseif obj:IsA("UIGradient") then
		appendProps(lines, var, obj, exportGradientProps)
	elseif obj:IsA("UIAspectRatioConstraint") then
		appendProps(lines, var, obj, exportAspectProps)
	elseif obj:IsA("UISizeConstraint") then
		appendProps(lines, var, obj, exportSizeProps)
	elseif obj:IsA("UIScale") then
		appendProps(lines, var, obj, exportScaleProps)
	elseif obj:IsA("UIListLayout") then
		appendProps(lines, var, obj, exportListProps)
	elseif obj:IsA("UIGridLayout") then
		appendProps(lines, var, obj, exportGridProps)
	elseif obj:IsA("UIPageLayout") then
		appendProps(lines, var, obj, exportPageProps)
	end

	table.insert(lines, var .. ".Parent = " .. parentExpr)
	table.insert(lines, "")

	for _, child in ipairs(obj:GetChildren()) do
		serializeTree(lines, child, var, vars, indexRef)
	end
end

local function encodeValue(v)
	local t = typeof(v)

	if t == "string" then return {T = "s", V = v} end
	if t == "number" then return {T = "n", V = v} end
	if t == "boolean" then return {T = "b", V = v} end
	if t == "Color3" then return {T = "c3", V = {v.R, v.G, v.B}} end
	if t == "UDim" then return {T = "ud", V = {v.Scale, v.Offset}} end
	if t == "UDim2" then return {T = "ud2", V = {v.X.Scale, v.X.Offset, v.Y.Scale, v.Y.Offset}} end
	if t == "Vector2" then return {T = "v2", V = {v.X, v.Y}} end
	if t == "Vector3" then return {T = "v3", V = {v.X, v.Y, v.Z}} end
	if t == "EnumItem" then return {T = "e", V = tostring(v)} end
	if t == "Rect" then return {T = "rect", V = {v.Min.X, v.Min.Y, v.Max.X, v.Max.Y}} end
	if t == "BrickColor" then return {T = "bc", V = v.Name} end

	return nil
end

local function decodeValue(data)
	if type(data) ~= "table" then return data end

	local t = data.T
	local v = data.V

	if t == "s" then return v end
	if t == "n" then return tonumber(v) end
	if t == "b" then return v == true end
	if t == "c3" then return Color3.new(v[1] or 0, v[2] or 0, v[3] or 0) end
	if t == "ud" then return UDim.new(v[1] or 0, v[2] or 0) end
	if t == "ud2" then return UDim2.new(v[1] or 0, v[2] or 0, v[3] or 0, v[4] or 0) end
	if t == "v2" then return Vector2.new(v[1] or 0, v[2] or 0) end
	if t == "v3" then return Vector3.new(v[1] or 0, v[2] or 0, v[3] or 0) end
	if t == "rect" then return Rect.new(v[1] or 0, v[2] or 0, v[3] or 0, v[4] or 0) end
	if t == "bc" then return BrickColor.new(tostring(v)) end

	if t == "e" then
		local enumType, enumName = splitEnumString(v)
		if enumType and enumName then
			local ok, enumItem = pcall(function() return Enum[enumType][enumName] end)
			if ok then return enumItem end
		end
	end

	return nil
end

local function appendEncodedProps(obj, props)
	local function add(list)
		for _, prop in ipairs(list) do
			if prop ~= "Name" then
				local value = safeGet(obj, prop)
				if value ~= nil then
					local encoded = encodeValue(value)
					if encoded then props[prop] = encoded end
				end
			end
		end
	end

	if obj:IsA("GuiObject") then
		add(exportGuiProps)

		if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
			add(exportTextProps)
		end

		if obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
			add(exportImageProps)
		end

		if obj:IsA("ScrollingFrame") then
			add(exportScrollProps)
		end

		if obj:IsA("VideoFrame") then
			add(exportVideoProps)
		end

		if obj:IsA("TextButton") or obj:IsA("ImageButton") then
			local interact = safeGet(obj, "Interactable")
			if interact ~= nil then
				props.Interactable = encodeValue(interact)
			end
		end
	elseif obj:IsA("UIStroke") then
		add(exportStrokeProps)
	elseif obj:IsA("UICorner") then
		add(exportCornerProps)
	elseif obj:IsA("UIPadding") then
		add(exportPaddingProps)
	elseif obj:IsA("UIGradient") then
		add(exportGradientProps)
	elseif obj:IsA("UIAspectRatioConstraint") then
		add(exportAspectProps)
	elseif obj:IsA("UISizeConstraint") then
		add(exportSizeProps)
	elseif obj:IsA("UIScale") then
		add(exportScaleProps)
	elseif obj:IsA("UIListLayout") then
		add(exportListProps)
	elseif obj:IsA("UIGridLayout") then
		add(exportGridProps)
	elseif obj:IsA("UIPageLayout") then
		add(exportPageProps)
	end
end

local function shouldSkipForSerialize(obj)
	if obj.Name == "NORLI_Selection" then return true end
	if obj == EmptyHint then return true end

	if not (
		obj:IsA("GuiObject") or obj:IsA("UICorner") or obj:IsA("UIStroke") or
		obj:IsA("UIGradient") or obj:IsA("UIPadding") or obj:IsA("UIAspectRatioConstraint") or
		obj:IsA("UISizeConstraint") or obj:IsA("UIScale") or obj:IsA("UIListLayout") or
		obj:IsA("UIGridLayout") or obj:IsA("UIPageLayout")
	) then
		return true
	end

	return false
end

local function encodeNode(obj)
	local node = {
		C = obj.ClassName,
		N = obj.Name,
		P = {},
		A = {},
		Ch = {},
	}

	appendEncodedProps(obj, node.P)

	if obj:IsA("GuiObject") then
		local locked = obj:GetAttribute("NORLI_Locked")
		if locked then node.A.L = true end
	end

	for _, child in ipairs(obj:GetChildren()) do
		if not shouldSkipForSerialize(child) then
			table.insert(node.Ch, encodeNode(child))
		end
	end

	return node
end

local function exportProjectJSON()
	local project = {
		V = 4,
		Root = {},
	}

	for _, child in ipairs(DesignRoot:GetChildren()) do
		if isUserElement(child) then
			table.insert(project.Root, encodeNode(child))
		end
	end

	local ok, json = pcall(function() return HttpService:JSONEncode(project) end)
	return ok and json or "-- ERRO AO GERAR JSON"
end

local function decodeNode(node, parent, forceNewIds)
	if type(node) ~= "table" then return nil end

	local ok, obj = pcall(Instance.new, node.C)
	if not ok or not obj then return nil end

	obj.Name = node.N or node.C

	for propName, encoded in pairs(node.P or {}) do
		local value = decodeValue(encoded)
		if value ~= nil then
			safeSet(obj, propName, value)
		end
	end

	if node.A and node.A.L then
		obj:SetAttribute("NORLI_Locked", true)
	end

	obj.Parent = parent

	for _, child in ipairs(node.Ch or {}) do
		decodeNode(child, obj, forceNewIds)
	end

	if obj:IsA("GuiObject") then
		if forceNewIds or not obj:GetAttribute("NORLI_ID") then
			obj:SetAttribute("NORLI_ID", newId())
		end
		connectSelectable(obj)
	end

	return obj
end

	local function rebuildRecent()
		for _, child in ipairs(recentGrid:GetChildren()) do
			if child:IsA("GuiButton") then child:Destroy() end
		end

		if #recentClasses == 0 then
			recentSection.Visible = false
			recentGrid.Visible = false
			recentGrid.Size = UDim2.new(1, 0, 0, 0)
			return
		end

		recentSection.Visible = true
		recentGrid.Visible = true
		recentGrid.Size = UDim2.new(1, 0, 0, 64)

		for _, className in ipairs(recentClasses) do
			local info = elementByClass[className]
			if info then
				elementCard(recentGrid, info)
			end
		end
	end

	local function rebuildGrid(filterText)
		for _, child in ipairs(grid:GetChildren()) do
			if child:IsA("GuiButton") then child:Destroy() end
		end

		filterText = tostring(filterText or ""):lower()
		local shown = 0

		for _, info in ipairs(elements) do
			if filterText == "" or info.label:lower():find(filterText, 1, true) or info.class:lower():find(filterText, 1, true) then
				elementCard(grid, info)
				shown += 1
			end
		end

	end

	rebuildGrid("")

local function openLayers()
	createDrawer("CAMADAS")

	refreshLayers = function()
		if currentTab ~= "LAYERS" or not panelBody then return end

		clearBody(panelBody)

		local items = {}

		local function scan(p, depth)
			for _, child in ipairs(p:GetChildren()) do
				if child:IsA("GuiObject") and child ~= EmptyHint and child.Name ~= "NORLI_Selection" then
					table.insert(items, {obj = child, depth = depth})
					scan(child, depth + 1)
				end
			end
		end

		scan(DesignRoot, 0)

		if #items == 0 then
			addTag(panelBody, "NENHUM ELEMENTO AINDA — vá em OBJETOS ou MODELOS")
			return
		end

		addTag(panelBody, tostring(#items) .. " elemento(s)")

		for _, item in ipairs(items) do
			layerRow(panelBody, item.obj, item.depth)
		end
	end

	refreshLayers()
end

local function openAssets()
	local body = createDrawer("ASSETS")

	addTag(body, "A pesquisa usa a rede do executor quando disponível.")

	local function applyAssetId(id)
		if not selected then
			toast("SELECIONE UM ELEMENTO", C.muted)
			return
		end

		id = tostring(id):gsub("%D", "")
		if id == "" then
			toast("ID INVÁLIDO", C.red)
			return
		end

		saveUndoPoint()

		if selected:IsA("ImageLabel") or selected:IsA("ImageButton") then
			selected.Image = "rbxassetid://" .. id
			toast("IMAGEM APLICADA", C.green)
		elseif selected:IsA("VideoFrame") then
			selected.Video = "rbxassetid://" .. id
			toast("VÍDEO APLICADO", C.green)
		else
			toast("SELECIONE IMAGEM OU VÍDEO", C.amber)
			return
		end

		table.insert(recentAssets, 1, id)
		refreshProperties()
	end

	section(body, "APLICAR POR ID")

	textBox(body, "ASSET ID", "", function(v)
		applyAssetId(v)
	end, "ex.: 123456789")

	section(body, "PESQUISAR CATÁLOGO")

	local search = textBox(body, "PESQUISA", "", function() end, "ex.: icon, button, bg")
	local status = addTag(body, "Aguardando pesquisa...")

	local results = make("Frame", {
		Size = UDim2.new(1, 0, 0, 300),
		BackgroundTransparency = 1,
		ZIndex = 202,
	}, body)

	local resultGrid = make("UIGridLayout", {
		CellSize = UDim2.new(0.48, 0, 0, 86),
		CellPadding = UDim2.new(0, 7, 0, 7),
		HorizontalAlignment = Enum.HorizontalAlignment.Center,
		SortOrder = Enum.SortOrder.LayoutOrder,
	}, results)

	local function doSearch(keyword)
		keyword = tostring(keyword or ""):gsub("^%s+", ""):gsub("%s+$", "")
		if keyword == "" then
			status.Text = "Digite uma palavra-chave."
			return
		end

		local req = getRequestFunction()
		if not req then
			status.Text = "Executor sem request/http_request."
			return
		end

		status.Text = "Pesquisando..."

		for _, child in ipairs(results:GetChildren()) do
			if child:IsA("GuiButton") then child:Destroy() end
		end

		local url = "https://catalog.roblox.com/v1/search/items/details?Keyword=" .. urlEncode(keyword) .. "&Limit=30&SortType=0&Category=1&Subcategory=8"

		task.spawn(function()
			local ok, response = pcall(function()
				return req({Url = url, Method = "GET"})
			end)

			if not ok or not response then
				status.Text = "Falha ao consultar o catálogo."
				return
			end

			local responseText = response.Body or response.body
			if type(responseText) ~= "string" then
				status.Text = "Resposta inválida."
				return
			end

			local okJson, data = pcall(function() return HttpService:JSONDecode(responseText) end)
			if not okJson or type(data) ~= "table" or type(data.data) ~= "table" then
				status.Text = "Nenhum resultado compatível."
				return
			end

			local count = 0

			for _, item in ipairs(data.data) do
				local id = item.id or item.assetId
				local name = item.name or "Asset"

				if id then
					count += 1

					local card = make("TextButton", {
						BackgroundColor3 = C.card,
						BorderSizePixel = 0,
						AutoButtonColor = false,
						Text = "",
						ZIndex = 204,
					}, results)
					corner(card, 9)

					make("TextLabel", {
						Size = UDim2.new(1, -10, 0, 34),
						Position = UDim2.new(0, 5, 0, 5),
						BackgroundTransparency = 1,
						Text = tostring(name),
						Font = Enum.Font.GothamBold,
						TextSize = 9,
						TextColor3 = C.text,
						TextWrapped = true,
						ZIndex = 205,
					}, card)

					make("TextLabel", {
						Size = UDim2.new(1, -10, 0, 22),
						Position = UDim2.new(0, 5, 0, 43),
						BackgroundTransparency = 1,
						Text = "ID " .. tostring(id),
						Font = Enum.Font.Code,
						TextSize = 8,
						TextColor3 = C.muted,
						ZIndex = 205,
					}, card)

					card.Activated:Connect(function()
						applyAssetId(id)
					end)
				end
			end

			results.Size = UDim2.new(1, 0, 0, math.max(300, math.ceil(count / 2) * 93))
			status.Text = count > 0 and (count .. " resultado(s). Toque para aplicar.") or "Nenhum resultado."
		end)
	end

	button(body, "🔎 PESQUISAR AGORA", function()
		doSearch(search.Text)
	end)
end

local function openProject()
	local body = createDrawer("PROJETO")

	section(body, "EXPORTAR PARA O JOGO")
	addTag(body, "Gera um script Lua para criar a interface fora do editor.")

	local exportBox = makeCodeBox(body, exportLua(), 170)

	button(body, "⧉ COPIAR SCRIPT LUA", function()

			toast("SCRIPT LUA COPIADO", C.green)
		else
			exportBox:CaptureFocus()
			toast("COPIE MANUALMENTE", C.muted)
		end
	end)

	button(body, "↻ ATUALIZAR LUA", function()
		exportBox.Text = exportLua()
		toast("LUA ATUALIZADO", C.accent)
	end)

	section(body, "SALVAR PROJETO")
	addTag(body, "Copia um JSON do projeto para colar depois.")

	local jsonBox = makeCodeBox(body, exportProjectJSON(), 130)

	button(body, "💾 COPIAR JSON", function()
		if safeClipboard(jsonBox.Text) then
			toast("JSON COPIADO", C.green)
		else
			jsonBox:CaptureFocus()
			toast("COPIE MANUALMENTE", C.muted)
		end
	end)

	button(body, "↻ ATUALIZAR JSON", function()
		jsonBox.Text = exportProjectJSON()
		toast("JSON ATUALIZADO", C.accent)
	end)

	section(body, "CARREGAR PROJETO")
	addTag(body, "Cole abaixo o JSON salvo anteriormente.")

	local importBox = textBox(body, "COLE O JSON AQUI", "", function() end, "cole aqui...", true)

	local armed = false
	local importBtn = button(body, "⤓ IMPORTAR PROJETO", function()
		if importBox.Text == "" then
			toast("COLE UM JSON PRIMEIRO", C.muted)
			return
		end

		if not armed then
			armed = true
			importBtn.Text = "⚠ TOQUE DE NOVO PARA CONFIRMAR"
			importBtn.BackgroundColor3 = C.amber

			task.delay(4, function()
				if importBtn.Parent then
					armed = false
					importBtn.Text = "⤓ IMPORTAR PROJETO"
					importBtn.BackgroundColor3 = C.card
				end
			end)
			return
		end

		armed = false
		importBtn.Text = "⤓ IMPORTAR PROJETO"
		importBtn.BackgroundColor3 = C.card

		importProjectJSON(importBox.Text)
	end)

	section(body, "NOVO PROJETO")

	local newArmed = false
	local newBtn = button(body, "🗑 LIMPAR CANVAS", function()
		if not newArmed then
			newArmed = true
			newBtn.Text = "⚠ TOQUE DE NOVO PARA CONFIRMAR"
			newBtn.BackgroundColor3 = C.amber

			task.delay(4, function()
				if newBtn.Parent then
					newArmed = false
					newBtn.Text = "🗑 LIMPAR CANVAS"
					newBtn.BackgroundColor3 = C.card
				end
			end)
			return
		end

		newArmed = false
		saveUndoPoint()
		clearCanvas()
		toast("CANVAS LIMPO", C.amber)
	end)
end
