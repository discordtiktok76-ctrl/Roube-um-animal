local function IsValidInstance(instance)
	if not instance then
		return false
	end

	local ok, valid = pcall(function()
		return instance.Parent ~= nil
	end)

	return ok and valid
end

local function GetPath(root, names)
	local current = root

	for _, name in ipairs(names) do
		if not current then
			return nil
		end

		current = current:FindFirstChild(name)
	end

	return current
end

local function NormalizeName(name)
	return tostring(name):lower():gsub("%s+", ""):gsub("%W", "")
end

local function GetFirstRenderable(instance)
	if not IsValidInstance(instance) then
		return nil
	end

	if instance:IsA("Player") then
		local character = instance.Character
		if character and IsValidInstance(character) then
			return character
		end

		return nil
	end

	if instance:IsA("ObjectValue") then
		return GetFirstRenderable(instance.Value)
	end

	if instance:IsA("Model") then
		return instance
	end

	if instance:IsA("BasePart") then
		return instance
	end

	if instance:IsA("Tool") then
		local handle = instance:FindFirstChild("Handle")

		if handle and handle:IsA("BasePart") then
			return handle
		end
	end

	local model = instance:FindFirstChildWhichIsA("Model")
	if model then
		return model
	end

	local part = instance:FindFirstChildWhichIsA("BasePart")
	if part then
		return part
	end

	local ok, result = pcall(function()
		for _, desc in ipairs(instance:GetDescendants()) do
			if desc:IsA("Model") or desc:IsA("BasePart") then
				return desc
			end
		end

		return nil
	end)

	if ok then
		return result
	end

	return nil
end

local function IsGeneratorName(name)
	local normal = NormalizeName(name)

	return normal == "generator" or normal:find("generator", 1, true) ~= nil
end

local function AddFolderTargets(folder, color, desired)
	if not folder then
		return
	end

	for _, child in ipairs(folder:GetChildren()) do
		local adornee = GetFirstRenderable(child)

		if adornee and IsValidInstance(adornee) then
			desired[adornee] = color
		end
	end
end

local function AddGeneratorTargets(desired)
	local exactFolder = GetPath(workspace, {"Map", "Ingame", "Map"})
	local root = nil

	if exactFolder then
		root = exactFolder
	else
		local ingameFolder = GetPath(workspace, {"Map", "Ingame"})

		if ingameFolder then
			root = ingameFolder
		else
			root = workspace:FindFirstChild("Map")
		end
	end

	if not root then
		return
	end

	pcall(function()
		for _, desc in ipairs(root:GetDescendants()) do
			if IsGeneratorName(desc.Name) then
				local adornee = GetFirstRenderable(desc)

				if adornee and IsValidInstance(adornee) then
					desired[adornee] = COLORS.GENERATORS
				end
			end
		end
	end)
end

local function MakeHighlight(adornee, color)
	local highlight = Instance.new("Highlight")
	highlight.Name = "NorliESP_Aura"
	highlight.Adornee = adornee
	highlight.FillColor = color
	highlight.OutlineColor = color
	highlight.FillTransparency = 0.7
	highlight.OutlineTransparency = 0
	highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	highlight.Parent = adornee

	return highlight
end

local function SyncESP()
	local desired = {}

	local playersFolder = workspace:FindFirstChild("Players")

	if playersFolder then
		if ESPStates.KILLERS then
			AddFolderTargets(playersFolder:FindFirstChild("Killers"), COLORS.KILLERS, desired)
		end

		if ESPStates.SURVIVORS then
			AddFolderTargets(playersFolder:FindFirstChild("Survivors"), COLORS.SURVIVORS, desired)
		end

		if ESPStates.SPECTATORS then
			AddFolderTargets(playersFolder:FindFirstChild("Spectating"), COLORS.SPECTATORS, desired)
		end
	end

	if ESPStates.GENERATORS then
		AddGeneratorTargets(desired)
	end

	for adornee, highlight in pairs(ESPCache) do
		local keep = desired[adornee] ~= nil and IsValidInstance(adornee) and highlight and highlight.Parent

		if not keep then
			pcall(function()
				if highlight then
					highlight:Destroy()
				end
			end)

			ESPCache[adornee] = nil
		end
	end

	for adornee, color in pairs(desired) do
		local highlight = ESPCache[adornee]

		if highlight and highlight.Parent then
			highlight.FillColor = color
			highlight.OutlineColor = color
			highlight.Enabled = true
		else
			local ok, newHighlight = pcall(MakeHighlight, adornee, color)

			if ok and newHighlight then
				ESPCache[adornee] = newHighlight
			end
		end
	end
end

local function GetItemConfig(key)
	for _, config in ipairs(ItemConfigs) do
		if config.Key == key then
			return config
		end
	end

	return nil
end

local function MatchKeywords(name, keywords)
	local normal = NormalizeName(name)

	if normal == "" then
		return false
	end

	for _, keyword in ipairs(keywords) do
		local normalKeyword = NormalizeName(keyword)

		if normalKeyword ~= "" then
			if normal == normalKeyword or normal:find(normalKeyword, 1, true) then
				return true
			end
		end
	end

	return false
end

local SCAN_LIMIT = 6000

local function ScanContainer(container, keywords, depth, scanned)
	if not container or depth > 7 or scanned.count > SCAN_LIMIT then
		return nil
	end

	local children
	local ok = pcall(function()
		children = container:GetChildren()
	end)

	if not ok or not children then
		return nil
	end

	for _, child in ipairs(children) do
		scanned.count = scanned.count + 1

		if scanned.count > SCAN_LIMIT then
			return nil
		end

		if child.Name ~= "Players"
			and child.Name ~= "Terrain"
			and child.Name ~= "Camera"
			and child.Name ~= "NorliHub" then

			if MatchKeywords(child.Name, keywords) then
				if GetFirstRenderable(child) then
					return child
				end
			end

			if child:IsA("Folder") or child:IsA("Model") then
				local found = ScanContainer(child, keywords, depth + 1, scanned)

				if found then
					return found
				end
			end
		end
	end

	return nil
end

local function FindItem(config)
	local exact = GetPath(workspace, config.Path)

	if exact and GetFirstRenderable(exact) then
		return exact
	end

	local ingame = GetPath(workspace, {"Map", "Ingame"})

	if ingame then
		local found = ScanContainer(ingame, config.Keywords, 0, {count = 0})

		if found then
			return found
		end
	end

	local map = workspace:FindFirstChild("Map")

	if map then
		local found = ScanContainer(map, config.Keywords, 0, {count = 0})

		if found then
			return found
		end
	end

	if not map then
		local found = ScanContainer(workspace, config.Keywords, 0, {count = 0})

		if found then
			return found
		end
	end

	return nil
end

local function GetCachedItem(key)
	local cached = ItemCache[key]

	if cached and IsValidInstance(cached) and GetFirstRenderable(cached) then
		return cached
	end

	if cached then
		ItemCache[key] = nil
	end

	return nil
end

local function RefreshItem(key, force)
	local cached = GetCachedItem(key)

	if cached and not force then
		return cached
	end

	local now = tick()

	if not force and ItemLastScan[key] and now - ItemLastScan[key] < 3 then
		return cached
	end

	local config = GetItemConfig(key)

	if not config then
		return nil
	end

	ItemLastScan[key] = now

	local found = FindItem(config)
	ItemCache[key] = found

	return found
end

local function AnchorAllParts(instance)
	pcall(function()
		if instance:IsA("BasePart") then
			instance.Anchored = true
		end

		for _, desc in ipairs(instance:GetDescendants()) do
			if desc:IsA("BasePart") then
				desc.Anchored = true
			end
		end
	end)
end

local function MoveInstanceToPlayer(instance)
	local character = LocalPlayer.Character

	if not character then
		return false, "Personagem não encontrado."
	end

	local rootPart = character:FindFirstChild("HumanoidRootPart")
		or character:FindFirstChild("UpperTorso")
		or character:FindFirstChild("Torso")

	if not rootPart or not rootPart:IsA("BasePart") then
		return false, "Parte raiz do personagem não encontrada."
	end

	local targetCFrame = rootPart.CFrame * CFrame.new(0, 1.2, -3)

	local ok, errorMessage = pcall(function()
		if instance:IsA("Model") then
			if not instance.PrimaryPart then
				local firstPart = GetFirstRenderable(instance)

				if firstPart and firstPart:IsA("BasePart") then
					instance.PrimaryPart = firstPart
				end
			end

			if instance.PrimaryPart then
				instance:PivotTo(targetCFrame)
			else
				error("Model sem parte válida.")
			end

		elseif instance:IsA("BasePart") then
			instance.CFrame = targetCFrame

		elseif instance:IsA("Tool") then
			local handle = instance:FindFirstChild("Handle")

			if handle and handle:IsA("BasePart") then
				handle.CFrame = targetCFrame
			else
				instance.Parent = character
			end

		else
			local model = instance:FindFirstChildWhichIsA("Model")

			if model then
				if not model.PrimaryPart then
					local firstPart = GetFirstRenderable(model)

					if firstPart and firstPart:IsA("BasePart") then
						model.PrimaryPart = firstPart
					end
				end

				if model.PrimaryPart then
					model:PivotTo(targetCFrame)
				else
					error("Model sem parte válida.")
				end
			else
				local parts = {}

				for _, desc in ipairs(instance:GetDescendants()) do
					if desc:IsA("BasePart") then
						table.insert(parts, desc)
					end
				end

				if #parts == 0 then
					error("Item sem parte válida.")
				end

				local primary = parts[1]
				local offset = targetCFrame * primary.CFrame:Inverse()

				for _, part in ipairs(parts) do
					part.CFrame = offset * part.CFrame
				end
			end
		end

		AnchorAllParts(instance)
	end)

	return ok, errorMessage
end

local dragging = false
local dragStart
local startPosition
local dragInput

local function UpdateDrag(input)
	local delta = input.Position - dragStart

	FloatingButton.Position = UDim2.new(
		startPosition.X.Scale,
		startPosition.X.Offset + delta.X,
		startPosition.Y.Scale,
		startPosition.Y.Offset + delta.Y
	)
end

FloatingButton.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPosition = FloatingButton.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

FloatingButton.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then

		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if input == dragInput and dragging then
		UpdateDrag(input)
	end
end)

--// STATUS / NOTIFY
local SelectedItemKey = nil
local DropdownOpen = false
local NotifyToken = 0

local function Notify(message, color)
	if not ItemStatus then
		return
	end

	NotifyToken = NotifyToken + 1
	local token = NotifyToken

	ItemStatus.Text = message
	ItemStatus.TextColor3 = color or Color3.fromRGB(255, 255, 255)

	task.delay(2.5, function()
		if token == NotifyToken then
			ItemStatus.Text = ""
		end
	end)
end

--// ATUALIZAR ITEM UI
local function UpdateDropdownButton()
	local config = SelectedItemKey and GetItemConfig(SelectedItemKey)

	if config then
		local item = GetCachedItem(SelectedItemKey)
		DropdownButton.Text = "ITEM: " .. config.Display .. (item and "" or " (não encontrado)")
	else
		DropdownButton.Text = "SELECIONAR ITEM"
	end

	DropdownButton.Text = DropdownButton.Text .. (DropdownOpen and "  [-]" or "  [+]")
end

local function UpdateOptions()
	for _, config in ipairs(ItemConfigs) do
		local button = ItemOptionButtons[config.Key]

		if button then
			local item = GetCachedItem(config.Key)
			local available = item ~= nil

			button.Text = config.Display .. (available and "" or " (não encontrado)")

			if available then
				button.TextColor3 = Color3.fromRGB(225, 225, 225)
			else
				button.TextColor3 = Color3.fromRGB(110, 110, 110)
			end

			if SelectedItemKey == config.Key and available then
				button.BackgroundColor3 = Color3.fromRGB(60, 210, 100)
				button.TextColor3 = Color3.fromRGB(0, 0, 0)
			else
				button.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
			end
		end
	end
end

local function UpdateGrabButton()
	local available = SelectedItemKey and GetCachedItem(SelectedItemKey) ~= nil

	if available then
		GrabButton.BackgroundColor3 = Color3.fromRGB(60, 210, 100)
		GrabButton.TextColor3 = Color3.fromRGB(0, 0, 0)
	else
		GrabButton.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
		GrabButton.TextColor3 = Color3.fromRGB(150, 150, 150)
	end
end

local function UpdateItemUI()
	UpdateDropdownButton()
	UpdateOptions()
	UpdateGrabButton()
end

--// EVENTOS ITEM
DropdownButton.Activated:Connect(function()
	DropdownOpen = not DropdownOpen
	DropdownList.Visible = DropdownOpen

	if DropdownOpen then
		for _, config in ipairs(ItemConfigs) do
			RefreshItem(config.Key, false)
		end
	end

	UpdateItemUI()
end)

for _, config in ipairs(ItemConfigs) do
	local button = ItemOptionButtons[config.Key]

	button.Activated:Connect(function()
		local item = RefreshItem(config.Key, true)

		if not item then
			Notify("Item não encontrado no mapa.", Color3.fromRGB(255, 90, 90))
			return
		end

		SelectedItemKey = config.Key
		DropdownOpen = false
		DropdownList.Visible = false

		UpdateItemUI()
	end)
end

GrabButton.Activated:Connect(function()
	if not SelectedItemKey then
		Notify("Selecione um item primeiro.", Color3.fromRGB(255, 220, 80))
		return
	end

	local item = RefreshItem(SelectedItemKey, true)

	if not item then
		Notify("Item não encontrado.", Color3.fromRGB(255, 90, 90))
		UpdateItemUI()
		return
	end

	local ok, errorMessage = MoveInstanceToPlayer(item)

	if ok then
		Notify("Item teleportado!", Color3.fromRGB(60, 210, 100))
	else
		Notify("Falha: " .. tostring(errorMessage), Color3.fromRGB(255, 90, 90))
	end

	UpdateItemUI()
end)

end

UpdateItemUI()
